#!/bin/sh
# This script runs the GSview help
# $1 is the help file name
# $2 is the topic name

if [ -z "$1" ]
then
  echo "Usage: $0 URL [TAG]"
  echo "  URL: The URL to open"
  echo "  TAG: The anchor at that url"
  exit 1
fi

if [ -z "$2" ]
then
  TAG=''
else
  TAG="#$2"
fi 

if netscape -remote openFile\($1$TAG\) ; then
    exit 0
else
    netscape $1$TAG
fi 
exit 0
